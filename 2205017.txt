+struct proc*
+pick_lottery_winner(void)
+{
+  struct proc *p;
+  int total = 0;
+
+  // calculate total current tickets.
+  
+  for(p = proc; p < &proc[NPROC]; p++){
+    acquire(&p->lock);

+       p->in_queue == 0){

+    }

+  int r = (krand() % total) + 1;
+  int sum = 0;
+
+  // Find the winner.

+    acquire(&p->lock);

+    if(p->state == RUNNABLE &&
+       p->in_queue == 0){
+
+      sum += p->tickets_current;
+
+      if(sum >= r){

+void
+tick_account(struct proc *p)
+{
+  struct proc *q;
+
+  
+  // Account the process that actually used this tick.
+   
+  acquire(&p->lock);
+
+  p->running_time++;
+  p->queue_ticks[p->in_queue]++;
+
+  release(&p->lock);
+
+  
+  // Every RUNNABLE process has now waited one more tick.
+ 
+  for(q = proc; q < &proc[NPROC]; q++){
+    acquire(&q->lock);
+
+    if(q->state == RUNNABLE)
+      q->waiting_time++;
+
+    release(&q->lock);
+  }

+void
+run_proc(struct cpu *c, struct proc *p)
+{
+  acquire(&p->lock);
+
+  if(p->state != RUNNABLE){
+    release(&p->lock);
+    return;
+  }
+
+  p->state = RUNNING;
+  c->proc = p;
+
+  p->times_scheduled++;
+
+  // It is no longer waiting
+  p->waiting_time = 0;
+
+  swtch(&c->context, &p->context);
+
+  c->proc = 0;
+  release(&p->lock);
+}

+uint64
+atoull(const char *s)
+{
+  uint64 n = 0;
+
+  while(*s >= '0' && *s <= '9'){
+    n = n * 10 + (*s - '0');
+    s++;
+  }
+
+  return n;
+}
+
+void
+dummy_work(uint64 iterations, int do_sleep)
+{
+  uint64 i;
+
+  for(i = 0; i < iterations; i++){
+
+    volatile uint64 x = i * i;
+    (void)x;
+
+    if(do_sleep &&
+       i != 0 &&
+       i % (iterations / 10 + 1) == 0){
+
+      sleep(1);
+    }
+  }
+}
+
+int
+main(int argc, char *argv[])
+{
+  if(argc != 3){
+    printf("usage: dummyproc <tickets> <iterations>\n");
+    exit(1);
+  }
+
+  int tickets = atoi(argv[1]);
+  uint64 iterations = atoull(argv[2]);
+
+  if(settickets(tickets) < 0){
+    printf("invalid ticket count\n");
+    exit(1);
+  }
+
+  int pid = fork();
+
+  if(pid < 0){
+    printf("fork failed\n");
+    exit(1);
+  }
+
+  if(pid == 0){
+    //voluntary promotion as child sleeps
+    dummy_work(iterations, 1);
+    exit(0);
+  }
+
+//parent is cpu bound
+  dummy_work(iterations, 0);
+
+  wait(0);
+
+  exit(0);
+}
