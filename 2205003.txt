+static uint64
+parse_u64(const char *s)
+{
+  uint64 n = 0;
+  if(*s == 0)
+    return 0;
+  for(; *s; s++){
+    if(*s < '0' || *s > '9')
+      return 0;
+    n = n * 10 + (*s - '0');
+  }
+  return n;
+}
+
+static int
+self_index(int pid)
+{
+  if(getpinfo(&ps) < 0)
+    return -1;
+  for(int i = 0; i < NPROC; i++)
+    if(ps.pid[i] == pid)
+      return i;
+  return -1;
+}
+
+static void
+busy_work(uint64 iters)
+{
+  volatile uint64 x = 0;
+  for(uint64 i = 0; i < iters; i++)
+    x = x * 31 + i;
+}
+
+static void
+sleepy_work(uint64 iters)
+{
+  volatile uint64 x = 0;
+  int me = getpid();
+  for(uint64 i = 0; i < iters; i++){
+    x = x * 31 + i;
+    if((i + 1) % CHUNK == 0){
+      int k = self_index(me);
+      if(k >= 0 && ps.inQ[k] == 1 && ps.running_time[k] >= TIME_LIMIT_1 / 2)
+        sleep(1);
+    }
+  }
+}

+int
+main(int argc, char *argv[])
+{
+  if(argc != 3){
+    fprintf(2, "usage: dummyproc <tickets> <iterations>\n");
+    exit(1);
+  }
+
+  int tickets = atoi(argv[1]);
+  uint64 iters = parse_u64(argv[2]);
+  if(iters == 0){
+    fprintf(2, "dummyproc: iterations must be a positive number\n");
+    exit(1);
+  }
+

+  }
